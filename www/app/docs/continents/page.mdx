### <T k="docs.common.enhancedFeatures" /> [#access-enhanced-features]

<T k="docs.continents.enhancedDesc" />

```ts filename="enhanced-continents.ts" showLineNumbers copy
import { Continents } from 'countries-geo';

const continent = Continents.getAll[0];

// Access ISO3 code
console.log(continent.iso3); // "AFR"

// Access coordinates
console.log(continent.mtd?.coor); // { lat: 0.0, lng: 20.0 }

// Access multi-language names
console.log(continent.ns.fr); // "Afrique"
console.log(continent.ns.es); // "África"
```
