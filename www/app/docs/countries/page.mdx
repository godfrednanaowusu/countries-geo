### <T k="docs.common.enhancedFeatures" /> [#access-enhanced-features]

<T k="docs.countries.enhancedDesc" />

```ts filename="enhanced-countries.ts" showLineNumbers copy
import { Countries } from 'countries-geo';

const country = Countries.getAll[0];

// Access ISO3 code
console.log(country.iso3); // "GHA"

// Access telephone code
console.log(country.mtd?.telc); // "+233"

// Access coordinates
console.log(country.mtd?.coor); // { lat: 8.1, lng: -1.0 }

// Access multi-language names
console.log(country.ns.fr); // "Ghana"
console.log(country.ns.es); // "Ghana"
```
