### <T k="docs.common.enhancedFeatures" /> [#access-enhanced-features]

<T k="docs.cities.enhancedDesc" />

```ts filename="enhanced-cities.ts" showLineNumbers copy
import { Cities } from 'countries-geo';

const city = Cities.getAll[0];

// Access coordinates
console.log(city.mtd?.coor); // { lat: 5.6, lng: -0.2 }

// Access multi-language names
console.log(city.ns.fr); // "Achimota"
```
