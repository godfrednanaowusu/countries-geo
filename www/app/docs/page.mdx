### <T k="docs.intro.backwardCompat" /> [#backward-compatibility]
<T k="docs.intro.backwardCompatDesc" />

```ts filename="enhanced-data.ts" showLineNumbers copy
import { Countries } from 'countries-geo';

// Legacy format (still works)
console.log(Countries.getAll[0]); 
// { iso: "GH", name: "Ghana" }

// Enhanced format
console.log(Countries.getAll[0]); 
// { 
//   iso: "GH", name: "Ghana",           // Legacy fields
//   iso2: "GH", iso3: "GHA",            // Enhanced ISO codes
//   ns: { en: "Ghana", fr: "Ghana" },  // Multi-language names
//   mtd: {                              // Metadata
//     telc: "+233",                     // Telephone code
//     coor: { lat: 8.1, lng: -1.0 }    // Coordinates
//   }
// }
```
